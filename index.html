Run this in your Supabase Dashboard → SQL Editor to create a test table with open read/write policies:

create table if not exists public.items (
  id bigint generated by default as identity primary key,
  name text not null,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

alter table public.items enable row level security;

create policy "Allow public select" on public.items for select using (true);
create policy "Allow public insert" on public.items for insert with check (true);
create policy "Allow public delete" on public.items for delete using (true);
